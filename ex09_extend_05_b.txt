// LED 0.5초 간격 점멸 (C/C++ 코드)

#include <Arduino.h>
const int LED = 6;

void setup() {
  pinMode(LED, OUTPUT);
}

void loop() {
  digitalWrite(LED, 1);
  delay(500);
  digitalWrite(LED, 0);
  delay(500);
}

int main() {
  init();
  setup();
  for (;;) {
    loop();
    if (serialEventRun) serialEventRun();
  }
  return 0;
}
