// LED 점멸할 때의 LED 상태 읽기

const int BUTTON = 2;
const int LED = 6;

void setup() {
  pinMode(BUTTON, INPUT);
  pinMode(LED, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  int value1 = digitalRead(BUTTON);	// 푸시버튼 상태값 읽기
  int value2 = digitalRead(LED);		// LED 상태값 읽기
  digitalWrite(LED, !value1);		// 푸시버튼을 누르면 LED ON
  Serial.print("LED State: ");
  Serial.println(value2);
}
