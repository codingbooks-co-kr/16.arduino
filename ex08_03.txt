// 가변저항 10비트 중 0번 비트 세트

const int VARIABLE_R = A1;

void setup() {
  Serial.begin(9600);
}

void loop() {
  int value = analogRead(VARIABLE_R);	// 가변저항값 읽기
  Serial.print(value, BIN);			// 10진수를 2진 표현으로 시리얼 모니터에 출력
  Serial.print(" ");
  value |= 0x1;				// value의 0번 비트를 1로 설정 (→비트 세트)
  Serial.println(value, BIN);			// 10진수를 2진 표현으로 시리얼 모니터에 출력
  delay(100);
}
