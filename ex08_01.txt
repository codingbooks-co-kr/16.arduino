// 10진수, 2진수, 16진수 값 출력

void setup() {
  Serial.begin(9600);
  int value = 12;

  Serial.print("int: ");
  Serial.println(value);		// 10진수 정수

  Serial.print("DEC: ");
  Serial.println(value, DEC);       	// 10진수 전송 (기본, DEC 생략 가능)

  Serial.print("BIN: ");
  Serial.println(value, BIN);   	// 10진수 값을 2진수 형식으로 시리얼 모니터에 출력

  Serial.print("HEX: ");
  Serial.println(value, HEX);     	// 10진수 값을 16진수 형식으로 시리얼 모니터에 출력
}

void loop() { }
