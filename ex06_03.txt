// 가변저항을 돌려 LED 순차 점등

const int VARIABLE_R = A1;
const int LED[] = {6, 7, 8, 12};		// 요소수는 4개이므로 LED[4]와 같은 의미

void setup() {
  pinMode(VARIABLE_R, INPUT);
  for (int i = 0; i < 4; i++) {		// LED 4개핀 모두 OUTPUT으로 설정
    pinMode(LED[i], OUTPUT);
  }
  Serial.begin(9600);
}

void loop() {
  int value = analogRead(VARIABLE_R);	// 가변저항값 읽기
  int num = map(value, 0, 1023, 0, 4);
  Serial.print("Variable Resistor: ");
  Serial.print(value);
  Serial.print(" : ");
  Serial.println(num);
  for (int i = 0; i < 4; i++) {   		// LED 전부 소등
    digitalWrite(LED[i], 0);
  }
  for (int i = 0; i < num; i++) { 		// LED 순차 점등 (num 값만큼 반복)
    digitalWrite(LED[i], 1);
  }
}
