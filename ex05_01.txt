// digitalWrite() 함수를 사용한 LED 밝기 조절 모방

const int LED = 6;

void setup() {
  pinMode(LED, OUTPUT);
}

void loop() {
  digitalWrite(LED, 0);		// 500msec 동안 LED 오프
  delay(500);
  for (int i = 0; i < 50; i++) {	// LED 온:오프 시간 비율 = 100msec : 400msec
    digitalWrite(LED, 1);
    delay(2);
    digitalWrite(LED, 0);
    delay(8);
  }
  for (int i = 0; i < 50; i++) {	// LED 온:오프 시간 비율 = 200msec : 300msec
    digitalWrite(LED, 1);
    delay(4);
    digitalWrite(LED, 0);
    delay(6);
  }
  for (int i = 0; i < 50; i++) {	// LED 온:오프 시간 비율 = 300msec : 200msec
    digitalWrite(LED, 1);
    delay(6);
    digitalWrite(LED, 0);
    delay(4);
  }
  for (int i = 0; i < 50; i++) {	// LED 온:오프 시간 비율 = 400msec : 100msec
    digitalWrite(LED, 1);
    delay(8);
    digitalWrite(LED, 0);
    delay(2);
  }
  digitalWrite(LED, 1);		// 500msec 동안 LED 온
  delay(500);
}
