// 어두울수록 LED 밝게 하기

const int LIGHT = A3;
const int LED = 6;
#define MIN 100		// 주위의 밝기에 따라 수정 요! (센서를 완전히 가린 상태)
#define MAX 300		// 주위의 밝기에 따라 수정 요! (센서를 가리지 않은 상태)

void setup() {
  pinMode(LIGHT, INPUT);
  pinMode(LED, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  int value = analogRead(LIGHT);
  Serial.print("LIGHT: ");				// 조도센서값을 시리얼 모니터로 전송
  Serial.print(value);
  value = constrain(value, MIN, MAX);		// value값제한: MIN∼MAX
  int brightness = map(value, MIN, MAX, 255, 0);	// value값변환: MIN∼MAX→255∼0
  analogWrite(LED, brightness);			// LED 밝기 제어
  Serial.print(" Brightness: ");
  Serial.println(brightness);				// 밝기값을 시리얼 모니터로 전송
  delay(100);
}
