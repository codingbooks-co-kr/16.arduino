// 센서 3개(1∼3번 핀) 값의 백분율 출력 (유형4)

void setup() {
  Serial.begin(9600);
  for (int i = 1; i <= 3; i++) {
    pinMode(i, INPUT);
  }
}

void loop() {
  for (int i = 1; i <= 3; i++) {
    int value = sensorPercent(i); 		// 함수 호출부 (유형4)
    Serial.print(" Pin");
    Serial.print(i);
    Serial.print(": ");
    Serial.print(value);
  }
  Serial.println();				// 개행: Serial.print('\n');와 같은 코드
  delay(100);
}

int sensorPercent(int pin) {      		// 함수 정의부 (유형4)
  int val = analogRead(pin);
  int percent = map(val, 0, 1023, 0, 100);
  return percent;				// 함수 반환값
}
