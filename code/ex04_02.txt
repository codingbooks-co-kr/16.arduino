// 푸시버튼 2개를 동시에 눌러 LED 점등

const int BUTTON1 = 2;
const int BUTTON2 = 4;
const int LED = 6;

void setup() {
  pinMode(BUTTON1, INPUT);
  pinMode(BUTTON2, INPUT);
  pinMode(LED, OUTPUT);
}

void loop() {
  int value1 = digitalRead(BUTTON1);
  int value2 = digitalRead(BUTTON2);

  if ((value1 == 0) && (value2 == 0)) {	// 푸시버튼 2개를 동시에 누른 경우
    digitalWrite(LED, 1);
  } else {				// 그렇지 않은 경우
    digitalWrite(LED, 0);
  }
}
