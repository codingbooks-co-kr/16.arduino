// 가변저항으로 LED 밝기 조절

const int VARIABLE_R = A1;
const int LED = 6;

void setup() {
  pinMode(VARIABLE_R, INPUT);
  pinMode(LED, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  int value = analogRead(VARIABLE_R);		// 가변저항값 읽기
  int brightness = map(value, 0, 1023, 0, 255);	// 가변저항값(0∼1023)→밝기(0∼255)
  analogWrite(LED, brightness);
  Serial.print("Brightness: ");
  Serial.println(brightness);		// LED 밝기값을 시리얼 모니터로 전송 후 출력
  delay(100);
}
