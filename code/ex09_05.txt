// 정수로 LED 점멸 시간 제어 (미완성)

const int LED = 6;

void setup() {
  pinMode(LED, OUTPUT);
  Serial.begin(9600);
  Serial.println("Please enter number(1-9)");
}

void loop() {
  if (Serial.available()) {		// 수신 데이터가 있을 때
    char c = Serial.read();		// 수신 데이터(문자) 읽기
    if (c >= '1' && c <= '9') {	// 수신 문자가 숫자문자인 ‘1’∼‘9’ 사이인 경우
      int time = (c - '0') * 100;	// '0'의 아스키코드는 48, 예) ‘1’-'0' = 49-48 = 1
      digitalWrite(LED, 1);
      delay(time);
      digitalWrite(LED, 0);
      delay(time);
      digitalWrite(LED, 1);
      delay(time);
      digitalWrite(LED, 0);
      delay(time);
    }
  }
}
