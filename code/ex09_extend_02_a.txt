// 수신된 문자열로 LED 밝기 제어

const int LED = 6;

void setup() {
  pinMode(LED, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  if (Serial.available()) {
    // "led"와 "\n" 사이의 숫자문자열 처리
    if (Serial.findUntil("led", "\n")) {
      int bright = Serial.parseInt();
      int value = constrain(bright, 0, 255);
      analogWrite(LED, value);
      Serial.println(value);
    }
  }
}
