// 폴링(polling) 방식
// 수신된 정수로 LED 점멸 제어

const int LED = 6;

void setup() {
  pinMode(LED, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  // 지속적으로 수신 여부 검사
  if (Serial.available()) {
    char value = Serial.read();  // 읽기
    if (value == '1') {
      digitalWrite(LED, 1);
    } else if (value == '0') {
      digitalWrite(LED, 0);
    }
  }
}
