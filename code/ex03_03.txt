// RGB LED를 1초 간격으로 R→G→B 순으로 점멸

const int RedLED = 11;
const int GreenLED = 5;
const int BlueLED = 3;

void setup() {
  pinMode(RedLED, OUTPUT);
  pinMode(GreenLED, OUTPUT);
  pinMode(BlueLED, OUTPUT);
}

void loop() {
  digitalWrite(RedLED, 1);		// Red: (R, G, B) = (1, 0, 0)
  digitalWrite(GreenLED, 0);
  digitalWrite(BlueLED, 0);
  delay(1000);

  digitalWrite(RedLED, 0);		// Green: (R, G, B) = (0, 1, 0)
  digitalWrite(GreenLED, 1);
  digitalWrite(BlueLED, 0);
  delay(1000);

  digitalWrite(RedLED, 0);		// Blue: (R, G, B) = (0, 0, 1)
  digitalWrite(GreenLED, 0);
  digitalWrite(BlueLED, 1);
  delay(1000);
}
